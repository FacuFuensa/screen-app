# Screen UX v2 — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Que la app se entienda sola: dos pestañas ("Yo" / la dupla), un editor que avisa qué va a pasar antes de tocar, pedidos que no se duplican y se pueden cancelar, pantallas que se actualizan solas y un recorrido guiado interactivo para la primera vez.

**Architecture:** La lógica nueva y testeable (vista previa del cambio, plan del recorrido, estado `cancelled`) va a `ScreenCore`. El servidor suma una RPC `cancel_request`. En la app, un `ActivityModel` compartido reemplaza la carga duplicada de pedidos/eventos; `HomeView` pasa a dos pestañas (`YoView`, `PartnerView`); el recorrido es un `TourController` + anclas por `anchorPreference` + un overlay con recorte que deja pasar los toques solo sobre el control iluminado.

**Tech Stack:** SwiftUI (iOS 17.4, modo Swift 5), XcodeGen, supabase-swift 2.55.1, Supabase (Postgres + pgTAP + Edge Functions Deno), GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-10-13-screen-ux-v2-design.md` (y las reglas de negocio de `docs/superpowers/specs/2026-10-12-screen-design.md`)

public enum PreviewTone: Equatable, Sendable {
    case appliesNow, needsApproval
}

public struct ChangePreviewNote: Equatable, Sendable {
    public let tone: PreviewTone
    public let text: String

    public init(tone: PreviewTone, text: String) {
        self.tone = tone
        self.text = text
    }
}

/// What saving an edited limit will do, shown live under the editor so nobody
/// taps "Guardar" without knowing it sends a request.
public enum ChangePreview {
    /// `old` is nil for a new limit; `partnerName` is nil when there is nobody to
    /// approve, in which case every change applies at once.
    public static func note(old: LimitConfig?, new: LimitConfig, partnerName: String?) -> ChangePreviewNote? {
        guard let old else { return ChangePreviewNote(tone: .appliesNow, text: "Se aplica al instante.") }
        switch LimitChange.classify(from: old, to: new) {
        case .unchanged:
            return nil
        case .tightening:
            return ChangePreviewNote(tone: .appliesNow, text: "Menos tiempo: se aplica al instante.")
        case .loosening:
            guard let partnerName else { return ChangePreviewNote(tone: .appliesNow, text: "Se aplica al instante.") }
            let what = LimitChange.removesApps(from: old, to: new) ? "Sacar apps" : "Más tiempo"
            let current = Formatting.limitSummary(dailyMinutes: old.dailyMinutes, window: old.window)
            return ChangePreviewNote(tone: .needsApproval,
                                     text: "\(what): \(partnerName) tiene que aprobarlo. Hasta entonces seguís con \(current).")
        }
    }

    public static func buttonTitle(for note: ChangePreviewNote?, partnerName: String?) -> String {
        note?.tone == .needsApproval ? "Pedirle a \(partnerName ?? "tu dupla")" : "Guardar"
    }
}

    /// left to teach (existing users never see it).
    public static func firstPending(authorized: Bool, groupCount: Int, paired: Bool) -> TourStep? {
        if !authorized { return .permissions }
        if groupCount == 0 { return .newLimit }
        if !paired { return .invite }

        TourStep(rawValue: step.rawValue + 1)

```sql
-- The requester can take back a request while it is still pending.
alter table public.requests drop constraint requests_status_check;
alter table public.requests add constraint requests_status_check
  check (status in ('pending', 'granted', 'denied', 'superseded', 'cancelled'));

create function public.cancel_request(p_request_id uuid) returns void
language plpgsql security definer set search_path = public as $$
declare r public.requests;
begin
  select * into r from requests where id = p_request_id for update;
  if not found or r.from_user is distinct from auth.uid() then raise exception 'not_found'; end if;
  if r.status <> 'pending' then raise exception 'not_pending'; end if;
  update requests set status = 'cancelled' where id = r.id;
end $$;

revoke execute on function public.cancel_request(uuid) from public, anon;
grant execute on function public.cancel_request(uuid) to authenticated;
```

`Screen/Services/ActivityModel.swift`:
```swift
import Foundation
import ScreenCore

/// Requests and events between the two people, shared by both tabs so they never
/// disagree, and the single place that answers or cancels (one call per tap).
@MainActor
final class ActivityModel: ObservableObject {
    @Published private(set) var requests: [RequestRow] = []
    @Published private(set) var events: [EventRow] = []
    @Published private(set) var busy: Set<UUID> = []
    @Published var error: String?

    let backend = Backend.shared

    func load() async {
        if Demo.isOn {
            requests = Demo.requests
            events = Demo.events
            return
        }
        do {
            async let loadedRequests = backend.requests()
            async let loadedEvents = backend.events()
            requests = try await loadedRequests
            events = try await loadedEvents
            error = nil
        } catch {
            self.error = error.localizedDescription
        }
    }

    func incoming(from partnerID: UUID?) -> [RequestRow] {
        guard let partnerID else { return [] }
        return requests.filter { $0.fromUser == partnerID && $0.effectiveStatus() == .pending }
    }

    func ownPendingExtra(me: UUID?) -> RequestRow? {
        requests.first { $0.fromUser == me && $0.kind == .extraTime && $0.effectiveStatus() == .pending }
    }

    func hasOwnPending(me: UUID?) -> Bool {
        requests.contains { $0.fromUser == me && $0.effectiveStatus() == .pending }
    }

    /// Repeated taps while the call is in flight are ignored.
    @discardableResult
    func answer(_ requestID: UUID, grant: Bool, minutes: Int? = nil) async -> Bool {
        guard !busy.contains(requestID) else { return false }
        busy.insert(requestID)
        defer { busy.remove(requestID) }
        if Demo.isOn {
            requests.removeAll { $0.id == requestID }
            return true
        }
        do {
            try await backend.answer(requestID: requestID, grant: grant, minutes: minutes)
            await backend.notify(requestID: requestID)
            await load()
            return true
        } catch {
            self.error = error.localizedDescription
            await load()
            return false
        }
    }

    func cancel(_ requestID: UUID) async {
        guard !busy.contains(requestID) else { return }
        busy.insert(requestID)
        defer { busy.remove(requestID) }
        if Demo.isOn {
            requests.removeAll { $0.id == requestID }
            return
        }
        do {
            try await backend.cancelRequest(requestID)
        } catch {
            self.error = error.localizedDescription
        }
        await load()
    }
}
```

import ScreenCore

/// Plain-Spanish summaries of a limit, shared by both tabs and the editor.
enum LimitText {
    static func summary(dailyMinutes: Int?, window: BlockWindow?, apps: Int) -> String {
        var parts: [String] = []
        if apps > 0 { parts.append(apps == 1 ? "1 app" : "\(apps) apps") }
        if let dailyMinutes { parts.append("\(Formatting.minutes(dailyMinutes)) por día") }
        if let window { parts.append("bloqueo \(Formatting.window(window))") }
        return parts.joined(separator: " · ")
    }

    static func summary(_ group: LimitGroup) -> String {
        summary(dailyMinutes: group.dailyMinutes, window: group.window, apps: group.appCount)
    }

    static func summary(_ group: GroupRow) -> String {
        summary(dailyMinutes: group.dailyMinutes, window: group.window, apps: group.appCount)
    }

    /// "45 min por día → 75 min por día", "borrar el límite" or "cambia qué apps entran".
    static func change(from old: LimitGroup, to edit: PendingEdit) -> String {
        if edit.delete { return "borrar el límite" }
        let before = Formatting.limitSummary(dailyMinutes: old.dailyMinutes, window: old.window)
        let after = Formatting.limitSummary(dailyMinutes: edit.group.dailyMinutes, window: edit.group.window)
        return before == after ? "cambia qué apps entran" : "\(before) → \(after)"
    }
}

import ScreenCore
import SwiftUI

/// Drives the first-run tour. Steps only advance through `complete(_:)`, called by
/// the real action (tapping the lit button, picking apps…), never by a "Next" button.
@MainActor
final class TourController: ObservableObject {
    static let shared = TourController()

    @Published private(set) var current: TourStep?
    @Published var showFinal = false

    private let defaults = UserDefaults.standard
    private var done: Bool {
        get { defaults.bool(forKey: "tourDone") }
        set { defaults.set(newValue, forKey: "tourDone") }
    }

    func start(authorized: Bool, groupCount: Int, paired: Bool) {
        guard !done, current == nil, !showFinal else { return }
        current = TourPlan.firstPending(authorized: authorized, groupCount: groupCount, paired: paired)
        if current == nil { done = true }
    }

    func complete(_ step: TourStep) {
        guard current == step else { return }
        withAnimation(.easeInOut(duration: 0.25)) { current = TourPlan.next(after: step) }
        if current == nil { finish() }
    }

    /// Pairing ends the tour wherever it is (the other person may have used my code).
    func pairingCompleted() {
        guard let step = current, step == .invite || step == .share else { return }
        current = nil
        finish()
    }

    func skip() {
        current = nil
        done = true
    }

    func restart() {
        done = false
        showFinal = false
        current = .newLimit
    }

    /// Demo screenshots only.
    func force(_ step: TourStep?, final: Bool = false) {
        current = step
        showFinal = final
    }

    private func finish() {
        done = true
        showFinal = true
    }
}

struct TourAnchorKey: PreferenceKey {
    static var defaultValue: [TourStep: Anchor<CGRect>] = [:]
    static func reduce(value: inout [TourStep: Anchor<CGRect>], nextValue: () -> [TourStep: Anchor<CGRect>]) {
        value.merge(nextValue()) { $1 }
    }
}

extension View {
    /// Marks the control the tour lights up for `step`.
    func tourAnchor(_ step: TourStep) -> some View {
        anchorPreference(key: TourAnchorKey.self, value: .bounds) { [step: $0] }
    }

    /// Draws the tour overlay for any anchor inside this view.
    func tourHost() -> some View { modifier(TourHost()) }
}

struct TourHost: ViewModifier {
    @ObservedObject private var tour = TourController.shared

    func body(content: Content) -> some View {
        content.overlayPreferenceValue(TourAnchorKey.self) { anchors in
            GeometryReader { proxy in
                if let step = tour.current, let anchor = anchors[step] {

                                onSkip: { tour.skip() },
                                onAccept: step == .minutes ? { tour.complete(.minutes) } : nil)
                }
            }
        }
    }
}

import ScreenCore
import SwiftUI

/// The whole area except the lit control, as one shape with a hole.
struct SpotlightShape: Shape {
    var hole: CGRect
    var cornerRadius: CGFloat

    func path(in rect: CGRect) -> Path {
        var path = Path(rect)
        path.addRoundedRect(in: hole, cornerSize: CGSize(width: cornerRadius, height: cornerRadius))
        return path
    }
}

/// Dims everything but the lit control. Touches inside the hole reach the real
/// control; touches anywhere else are swallowed so nobody wanders off.
struct TourOverlay: View {
    let step: TourStep
    let hole: CGRect
    let container: CGSize
    let onSkip: () -> Void
    let onAccept: (() -> Void)?

    var body: some View {
        let lit = hole.insetBy(dx: -8, dy: -8)
        let below = lit.maxY + 190 < container.height
        ZStack(alignment: .topLeading) {
            SpotlightShape(hole: lit, cornerRadius: 16)
                .fill(Color.black.opacity(0.6), style: FillStyle(eoFill: true))
                .contentShape(SpotlightShape(hole: lit, cornerRadius: 16), eoFill: true)
                .onTapGesture {}
            RoundedRectangle(cornerRadius: 16, style: .continuous)
                .stroke(Theme.amber, lineWidth: 3)
                .frame(width: lit.width, height: lit.height)
                .position(x: lit.midX, y: lit.midY)
                .allowsHitTesting(false)
            TourBubble(step: step, onSkip: onSkip, onAccept: onAccept)
                .frame(width: min(container.width - 32, 380))
                .position(x: container.width / 2, y: below ? lit.maxY + 95 : max(95, lit.minY - 95))
        }
        .frame(width: container.width, height: container.height)
        .transition(.opacity)
    }
}

struct TourBubble: View {
    let step: TourStep
    let onSkip: () -> Void
    let onAccept: (() -> Void)?

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text(TourCopy.title(step)).font(.headline).foregroundStyle(Theme.ink)
            Text(TourCopy.message(step)).font(.subheadline).foregroundStyle(Theme.ink.opacity(0.85))

                Text("Paso \(step.number) de \(TourStep.total)").font(.caption).foregroundStyle(.secondary)
                Spacer()
                if let onAccept {
                    Button("Así está bien", action: onAccept).font(.caption.bold())
                }
                Button("Saltear", action: onSkip).font(.caption)
            }
        }
        .padding(14)
        .background(Color.white, in: RoundedRectangle(cornerRadius: 16, style: .continuous))
        .shadow(color: .black.opacity(0.25), radius: 14, y: 6)
    }
}
